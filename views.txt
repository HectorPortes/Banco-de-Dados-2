-- Alunos
CREATE TABLE aluno (
    id_aluno     SERIAL PRIMARY KEY,
    nome_aluno   VARCHAR(250) NOT NULL,
    mae          VARCHAR(250),
    pai          VARCHAR(250),
    data_nascimento DATE,
    descricao    TEXT
);

-- Cursos
CREATE TABLE curso (
    id_curso    SERIAL PRIMARY KEY,
    nome_curso  VARCHAR(250) NOT NULL
);

-- Turmas
CREATE TABLE turma (
    id_turma    SERIAL PRIMARY KEY,
    nome_turma  VARCHAR(250) NOT NULL,
    data_inicio DATE,
    data_fim    DATE,
    id_curso    INT REFERENCES curso(id_curso)
);

-- Matrículas (relaciona aluno + turma)
CREATE TABLE matricula (
    id_matricula SERIAL PRIMARY KEY,
    id_aluno     INT REFERENCES aluno(id_aluno),
    id_turma     INT REFERENCES turma(id_turma),
    nota_final   NUMERIC(4,2),
    data_matricula DATE DEFAULT CURRENT_DATE
);

-- view para alunos matriculados em algum curso
CREATE OR REPLACE VIEW vw_alunos_matriculados AS
	SELECT a.id_aluno,
		a.nome_aluno,
		t.nome_turma,
		c.nome_curso,
		m.nota_final,
		m.data_matricula
	From aluno a
	JOIN matricula m ON m.id_aluno = a.id_aluno
	JOIN turma t ON t.id_turma = m.id_turma
	JOIN curso c ON c.id_curso = t.id_curso

SELECT * FROM vw_alunos_matriculados

-- view para buscar alunos aprovados
CREATE OR REPLACE VIEW vw_alunos_aprovados AS
	SELECT nome_aluno,
		nome_curso,
		nome_turma,
		nota_final
	FROM vw_alunos_matriculados
	WHERE nota_final >= 7

SELECT * FROM vw_alunos_aprovados ORDER BY nota_final DESC

SELECT
	aluno.nome_aluno,
	matricula.nota_final,
	CASE
		WHEN matricula.nota_final >= 7 THEN 'Aprovado'
		WHEN matricula.nota_final < 7 THEN 'Recuperaçâo'
		ELSE 'Reprovado'
	END AS situacao
FROM matricula
JOIN aluno ON aluno.id_aluno = matricula.id_aluno


SELECT
	a.nome_aluno,
	m.nota_final,
	CASE
		WHEN m.nota_final >= 9 THEN 'A'
		WHEN m.nota_final >= 7 THEN 'B'
		WHEN m.nota_final >= 5 THEN 'C'
		ELSE 'D'
	END AS conceito
FROM matricula m
JOIN aluno a ON a.id_aluno = m.id_aluno


SELECT
	a.nome_aluno,
	c.nome_curso,
	CASE c.id_curso
		WHEN 1 THEN 'Informatica'
		WHEN 2 THEN 'Edificaçôes'
		WHEN 3 THEN 'TSI'
		WHEN 4 THEN 'TRC'
		WHEN 5 THEN 'Eng Civil'
		ELSE 'Outro'
	END AS area
FROM matricula m
JOIN aluno a ON a.id_aluno = m.id_aluno
JOIN turma t ON t.id_turma = m.id_turma
JOIN curso c ON c.id_curso = t.id_curso
